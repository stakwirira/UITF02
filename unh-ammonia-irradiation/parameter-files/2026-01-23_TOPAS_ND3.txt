# 4. MATERIAL DEFINITIONS
# ------------------------------------------------------------------------------

# --- Frozen Deuterated Ammonia (ND3) ---
# ==========================================
# Material Definition: Solid Deuterated Ammonia (ND3)
# ==========================================

# 1. Define Deuterium Element (NIST Standard)
# Using Symbol "H" to satisfy TOPAS/Geant4 symbol requirements
# while setting AtomicMass to the Deuterium value.
s:El/Deuterium/Symbol     = "H"
i:El/Deuterium/Z          = 1
d:El/Deuterium/AtomicMass = 2.01410178 g/mole

# 2. Define the Final Material
# Mass Calculation:
# Nitrogen (14.0067) + 3*Deuterium (2.01410178 * 3) = 20.04900534
# Nitrogen Fraction: 14.0067 / 20.0490 = 0.698623
# Deuterium Fraction: 6.042305 / 20.0490 = 0.301377

sv:Ma/SolidND3/Components = 2 "Nitrogen" "Deuterium"
uv:Ma/SolidND3/Fractions  = 2 0.698623 0.301377
d:Ma/SolidND3/Density     = 1.007 g/cm3
s:Ma/SolidND3/DefaultColor = "skyblue"

# --- Liquid Argon (LAr) ---
sv:Ma/LiquidArgon/Components         = 1 "Argon"
uv:Ma/LiquidArgon/Fractions          = 1 1.0
d:Ma/LiquidArgon/Density             = 1.39 g/cm3
